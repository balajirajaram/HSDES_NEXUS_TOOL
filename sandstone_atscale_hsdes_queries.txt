SELECT id, title, status, status_reason, priority, suspect_area,
       platform, product_found, submitted_date
WHERE  tenant = 'server_platf' AND subject = 'bug'
  AND (title             CONTAINS 'sandstone'
       OR execution_summary CONTAINS 'sandstone'
       OR description        CONTAINS 'sandstone')

Q2 — CONFIRMED-SILICON slice (the "117")
      non-rejected server_platf.bug where suspect_area = silicon
      NOTE: suspect_area is an owner-assigned TRIAGE field. It captures
      *suspect* silicon; it does NOT itself require a bugeco/errata link.
------------------------------------------------------------------------
SELECT id, title, status, status_reason, priority, suspect_area,
       platform, product_found, submitted_date
WHERE  tenant = 'server_platf' AND subject = 'bug'
  AND  status != 'rejected'
  AND (title CONTAINS 'sandstone'
       OR execution_summary CONTAINS 'sandstone'
       OR description CONTAINS 'sandstone')
  AND  suspect_area = 'silicon'
SORTBY submitted_date DESC

NOTES ON EQL SYNTAX
